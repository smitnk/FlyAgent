# Host (Linux) build of the same native code the app uses, for CI tests.
cmake_minimum_required(VERSION 3.22)
project(flyagent_host CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
include(FetchContent)
foreach(o LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_CURL GGML_OPENMP)
  set(${o} OFF CACHE BOOL "" FORCE)
endforeach()
# same runtime CPU-variant loading as the app
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
if(DEFINED LLAMA_SRC)
  add_subdirectory(${LLAMA_SRC} llama)
else()
  FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b6999 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(llama)
endif()
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../app/src/main/cpp)
add_executable(engine_test engine_test.cpp ${SRC}/connectome.cpp)
add_executable(sim_cli sim_cli.cpp ${SRC}/connectome.cpp)
add_executable(llm_test llm_test.cpp ${SRC}/llm_core.cpp)
target_link_libraries(llm_test llama)
